Hi, I'm Srishti — a Rust engineer and Web3 security researcher who builds blockchain infrastructure and breaks DeFi protocols (responsibly).

// main.rs snippet
// Detect dangerous opcodes in bytecode
fn analyze_bytecode(bytes: &[u8]) -> Vec<Finding> {
  let mut findings = Vec::new();
  for (i, &op) in bytes.iter().enumerate() {
    match op {
      0xf4 => findings.push(Finding::DelegateCall(i)),
      0xff => findings.push(Finding::SelfDestruct(i)),
      0xf1 => findings.push(Finding::ExternalCall(i)),
      _ => {}
    }
  }
  findings
}

fn execute_call(ctx: &mut Context) {

  let mut pc = 0usize;

    ctx.stack.execute(op)?;
